```clojure
(mx/defn exclude-shared :- law/ExcludeSharedContract
  "Returns a new vector keeping only elements from `source` that do not exist in `exclusions`."
  [source exclusions]
  (let [exclusion-set (set exclusions)]
    (filterv #(not (contains? exclusion-set %)) source)))
```

### The Category: Store Protocol
First, define the legal transitions for the persistence layer. We use a Protocol because it is highly optimized and defines a strict boundary.

```clojure
(ns infra.mongo.store)

;; The Category of our persistence mechanism.
;; Implementations are expected to handle their own effects but honor this signature.
(defprotocol IStore
  (-insert [this doc] "Inserts a document into the store")
  (-find   [this query] "Finds documents matching the query"))

;; A Record to fulfill the protocol.
;; This represents a specific instantiated MongoDB collection.
(defrecord MongoCollection [collection-name schema-validator db-client]
  IStore
  (-insert [_ doc]
    (let [valid? (schema-validator doc)]
      ;; The Contract guard prevents illegal state before I/O
      (when-not valid?
        (throw (ex-info "Document failed schema validation" {:doc doc})))
      
      (^:async fn []
        (let [col (.collection db-client collection-name)
              res (await (.insertOne col (clj->js doc)))]
          (js->clj res :keywordize-keys true)))))
  
  (-find [_ query]
    (^:async fn []
      (let [col (.collection db-client collection-name)
            res (await (-> col
                              (.find (clj->js query))
                              (.toArray)))]
        (js->clj res :keywordize-keys true)))))
```

### The Contract: Schema and Validation
Next, define the contract guard. Malli is ideal here because schemas are just data, matching your DSL requirements. [slideshare](https://www.slideshare.net/slideshow/malli-inside-datadriven-schemas/229446966)

```clojure
(ns law.store
  (:require [malli.core :as m]))

;; A function that compiles your data-DSL `:store/schema`
;; into an executable contract validator.
(defn compile-schema-guard [schema-data]
  (let [compiled-schema (m/schema schema-data)]
    (fn [doc]
      (m/validate compiled-schema doc))))
```

### The Registry / Interpreter
Now, we write the system component that reads your data structure and dynamically provisions the resources. By separating the DSL from execution, we do not require a `def-something` macro per file; we map over the sequence of `:resources`.

```clojure
(ns domain.ussyverse.registry
  (:require [law.store :as law]
            [infra.mongo.store :as mongo]))

(defn register-store! 
  "Takes a store definition from the DSL and an active DB client, 
   returning an instantiated MongoCollection record."
  [{:store/keys [id schema]} db-client]
  ;; We strip namespaces using the `name` function to get the raw collection name.
  (let [collection-name (name id)
        validator       (law/compile-schema-guard schema)]
    ;; Returns a verified Fact: an instantiated record
    (mongo/->MongoCollection collection-name validator db-client)))

(defn build-scope 
  "Parses the resources and wires the dependencies."
  [resources db-client]
  (reduce (fn [acc resource]
            (cond
              ;; If it has store keys, compile it into our scope
              (:store/id resource)
              (assoc acc (:store/id resource) 
                     (register-store! resource db-client))
              
              ;; ... handle action or trigger interpretations here ...
              :else acc))
          {}
          resources))
```
